## 1. Install

```bash
pip install starlette uvicorn searchfn datafn
```

## 2. App

```python
from starlette.applications import Starlette
from starlette.routing import Route
from starlette.requests import Request
from starlette.responses import JSONResponse

from searchfn import create_searchfn_server
from datafn.db import PostgresAdapter
from schema import schema

db = PostgresAdapter(connection_string=os.environ["SEARCH_DB_URL"])

search_plugin = create_searchfn_server({
    "schema": schema,
    "db": db,
})

def wrap(handler):
    async def endpoint(request: Request):
        body = await request.json()
        return JSONResponse(await handler(body))
    return endpoint

routes = [Route(path, wrap(handler), methods=["POST"]) for path, handler in search_plugin["routes"].items()]

app = Starlette(routes=routes)
```

## 3. Run

```bash
uvicorn app:app --reload
```

## Smoke test
